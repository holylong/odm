cmake_minimum_required(VERSION 3.21)
project(ODM VERSION 0.1 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The WinHTTP/Winsock backend requires Windows")
endif()
add_library(odm_core core.cpp util.cpp net.cpp ftp.cpp mms.cpp rtmp.cpp)
target_compile_features(odm_core PUBLIC cxx_std_17)
target_compile_definitions(odm_core PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 SECURITY_WIN32)
target_link_libraries(odm_core PRIVATE winhttp ws2_32 secur32 crypt32)
option(ODM_GUI "Build the Qt desktop application" ON)
if(ODM_GUI)
  find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets)
  qt_add_executable(odm_desktop WIN32 desktop.cpp app.rc)
  qt_add_resources(odm_desktop "appicon" PREFIX "/" FILES app.ico)
  target_link_libraries(odm_desktop PRIVATE odm_core Qt6::Widgets)
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(odm_core_test test.cpp)
  target_link_libraries(odm_core_test PRIVATE odm_core)
  add_test(NAME download COMMAND Python3::Interpreter ${CMAKE_CURRENT_SOURCE_DIR}/test_download.py $<TARGET_FILE:odm_core_test>)
  set_tests_properties(download PROPERTIES TIMEOUT 60)
  add_test(NAME stream COMMAND Python3::Interpreter ${CMAKE_CURRENT_SOURCE_DIR}/test_stream.py $<TARGET_FILE:odm_core_test>)
  set_tests_properties(stream PROPERTIES TIMEOUT 240)
endif()
